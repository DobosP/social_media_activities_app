func notificationInboxQANormalize(t testing.TB,w *httptest.ResponseRecorder,replacement string)([]byte,map[string]string,map[string]any) {
	t.Helper();raw:=w.Body.Bytes();policy:=w.Header().Get("Content-Security-Policy-Report-Only");mode:="Content-Security-Policy-Report-Only"

	match:=regexp.MustCompile(`'nonce-([^']+)'`).FindStringSubmatch(policy)
	if w.Code!=http.StatusOK||w.Header().Get("Content-Type")!="text/html; charset=utf-8"||w.Header().Get("Cache-Control")!="private, no-cache"||!strings.Contains(w.Header().Get("Vary"),"Cookie")||len(w.Result().Cookies())!=0||len(match)!=2||match[1]=="" { t.Fatal("actual renderer private response/nonce shape differs") }

	if count<2||strings.Count(policy,"'nonce-"+nonce+"'")!=1{t.Fatal("actual CSP and HTML nonce disagree")}
	body:=bytes.ReplaceAll(raw,[]byte(attribute),[]byte(`nonce="`+replacement+`"`))
	if bytes.Contains(body,[]byte(nonce)){t.Fatal("nonce escaped admitted attribute context")}
	headers:=map[string]string{};replacements:=0
	for key,values:=range w.Header(){
		if len(values)!=1{t.Fatal("single-value renderer header required",key)};value:=values[0]

	}

	return body,headers,map[string]any{"raw_html_sha256":groupDetailQAHash(raw),"normalized_html_sha256":groupDetailQAHash(body),"attribute_replacements":count,"csp_replacements":replacements}
}
