type accountCenterQABridgeProof struct {
	Producer string `json:"producer_source"`

	Product []string `json:"product_paths"`
	Paths []string `json:"source_paths"`
}

func accountCenterQABridge(t testing.TB, root string) accountCenterQABridgeProof {
	t.Helper()
	raw, err := exec.Command("python3", "-I", filepath.Join(root, "tools/gui-native-account-center/qualify.py"), "--repo", root, "--producer-source", *accountCenterQAProducer, "--check-bridge").Output()
	var proof accountCenterQABridgeProof
	if err != nil || json.Unmarshal(raw, &proof) != nil || proof.Producer != *accountCenterQAProducer || !regexp.MustCompile(`^[0-9a-f]{40}$`).MatchString(proof.Head) || len(proof.Product) < 2000 || len(proof.Paths) != len(proof.Product)+5 { t.Fatal("exact committed full source/five-method bridge required", err) }
	return proof
}

	if err := os.Mkdir(filepath.Join(output, "notification-inbox"), 0700); err != nil { t.Fatal(err) }
	for _, mode := range notificationInboxQAModes {
		request, data := notificationInboxFixture(mode); request.Header.Set("Accept-Language", data["language"].(string)); data["csrf"] = "synthetic-notification-inbox-ui-csrf"; delete(data, "request")
		w := httptest.NewRecorder(); if err := r.Render(w, request, "web/notifications.html", data); err != nil { t.Fatal(err) }
		body, _, _ := notificationInboxQANormalize(t, w, "synthetic-notification-inbox-ui-nonce")
		path := "notification-inbox/"+mode+".html"; groupDetailQAWrite(t, filepath.Join(output, path), body)
		rows = append(rows, map[string]any{"kind": "notification-inbox", "name": mode, "path": path, "bytes": len(body), "sha256": groupDetailQAHash(body), "equal": false})
	}
	if len(rows) != 37 { t.Fatal("exact37 prior body candidates required; Python compares literal physical keeper bytes") }; return rows
}

func TestSocialAccountCenterQAExport(t *testing.T) {
	root, err := filepath.Abs("../../../.."); if err != nil { t.Fatal(err) }; bridge := accountCenterQABridge(t, root); before := groupDetailQASources(t, root, bridge.Paths)
	output := *accountCenterQAOutput; if !filepath.IsAbs(output) || !strings.Contains(output, "/_temp/") { t.Fatal("owned scratch output required") }; parent, err := filepath.EvalSymlinks(filepath.Dir(output)); if err != nil || parent != filepath.Dir(output) { t.Fatal("unalias output parent required") }; if err = os.Mkdir(output, 0700); err != nil { t.Fatal(err) }
	r := NewRenderer(root); states, normalizations := []map[string]any{}, []map[string]any{}
	for _, mode := range accountCenterQAModes {
		request, name, data := accountCenterQAFixture(t, mode); expected := accountCenterQAExpected(t, r, request, name, data); semanticBefore, err := json.Marshal(data); if err != nil { t.Fatal(err) }
		w := httptest.NewRecorder(); if err = r.Render(w, request, name, data); err != nil { t.Fatal(mode, err) }; body, headers, normalized := notificationInboxQANormalize(t, w, "synthetic-account-center-ui-nonce"); normalized["name"] = mode
		semanticAfter, err := json.Marshal(data); if err != nil || !bytes.Equal(semanticBefore, semanticAfter) { t.Fatal("renderer mutated synthetic source") }
		if bytes.Contains(body, []byte("PRIVATE-")) || bytes.Contains(body, []byte("<style")) || bytes.Contains(body, []byte("<script>notice")) || bytes.Contains(body, []byte("<script>label")) { t.Fatal("private or active fixture text leaked") }
		groupDetailQAWrite(t, filepath.Join(output, mode+".html"), body)
		states = append(states, map[string]any{"name": mode, "language": data["language"], "theme": data["display_theme"], "text": data["display_text"], "motion": data["display_motion"], "bytes": len(body), "sha256": groupDetailQAHash(body), "go_gzip_level9_bytes": notificationInboxQAGzip(t, body), "canonical": expected["canonical"], "synthetic_csrf": data["csrf"], "renderer_headers": headers, "expected": expected, "synthetic_source_sha256": groupDetailQAHash(semanticBefore)})
		normalizations = append(normalizations, normalized)
	}
	prior := accountCenterQAPrior(t, root, output, r); after := groupDetailQASources(t, root, bridge.Paths); left, _ := json.Marshal(before); right, _ := json.Marshal(after); if !bytes.Equal(left, right) || accountCenterQABridge(t, root).Head != bridge.Head { t.Fatal("source/Git changed during export") }
	manifest := map[string]any{"schema": 1, "PG_producer_source": *accountCenterQAProducer, "UI_method_source": bridge.Head, "source_unchanged": true, "sources_before": before, "sources_after": after, "states": states, "previous_body_candidates": prior, "normalization": map[string]any{"schema": 1, "kind": "render-nonce-only", "replacement": "synthetic-account-center-ui-nonce", "states": normalizations}, "baseline_keeper_sha256": "bc10a80ae1a9b0f404e9434f2a0065d941a0cb4a336d5ea06eb39032da09c7ea", "scope": "Eight fixed synthetic AccountView display fixtures only. Prior37 bodies require separate physical keeper equality. No genuine token/session/export/evidence input; no action submitted; no actual authorization or language-save claim."}
	raw, err := json.MarshalIndent(manifest, "", "  "); if err != nil { t.Fatal(err) }; groupDetailQAWrite(t, filepath.Join(output, "fixtures.json"), append(raw, '\n'))
}
